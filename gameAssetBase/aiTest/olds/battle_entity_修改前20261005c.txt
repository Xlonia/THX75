const PP_CAP := 500                ## 每角色默认最大P点500
const SPELL_CARRY_MAX := 16        ## 每角色每次决斗最多携带16张
var spell_cards := 2               ## 开局携带符卡（暂定2，待作者定数）

var spell_receive_cap := 5         ## 默认最多接受5张（可被存档/难度覆盖）

func can_receive_spell() -> bool:
	if dead:
		return false ## 满目疮痍/撤退者完全不参与接收
	return spell_received < spell_receive_cap and spell_cards < SPELL_CARRY_MAX

func receive_spell() -> void:
	spell_cards += 1
	spell_received += 1

## —— 能力/符卡屏蔽圆（策划：能力80px·0.5s起 / 次级符卡200px·1s起 / 主要符卡全屏·5s起）——
var shield_radius := 0.0        ## 0=无屏蔽（9999=全屏）
var shield_frames := 0

func apply_shield(radius: float, seconds: float) -> void:
	shield_radius = maxf(shield_radius, radius)
	shield_frames = maxi(shield_frames, int(seconds * 60.0))

func shield_active() -> bool:
	return shield_frames > 0 and shield_radius > 0.0

## 加P点（溢出由战场侧再分配）；返回溢出量
func add_pp(amount: int) -> int:
	pp += amount
	var overflow := pp - PP_CAP
	if overflow > 0:
		pp = PP_CAP
	else:
		overflow = 0
	return overflow

## 移动（角色：慢机/快机；机动折损→20%；杂兵：世界坐标由跟随+位移合成）
## dist_scale：时间尺度地基（批复 2026-10-01#8：月之怀表/须臾枝/时之魔珠类道具
## 要求"敌方速度、行进、AI、态势、已释放弹幕整体同步缩放"，不可只慢移动）。
## ②期先接 敌AI 移动与火控 delta；弹幕逐侧 dt 缩放随道具系统实装。
func move(dir: Vector2, dist_scale: float = 1.0) -> void:
	if dead or dir == Vector2.ZERO:
		return
	var spd := cruise_speed() if cruise_active and move_ok() else normal_speed()
	if not move_ok():
		spd = normal_speed() * 0.2 ## 机动折损：慢机20%
	spd *= dist_scale
	if kind == Kind.PRIVATE:
		## 杂兵位置=角色位置+相对位移（角色动它们跟着动）
		local_pos += dir.normalized() * spd
		sync_private_pos()
		return
	position += dir.normalized() * spd
	position = position.clamp(Vector2.ZERO, Vector2(400, 400))

## 杂兵世界坐标同步（owner 移动即跟随）
func sync_private_pos() -> void:
	if owner_ent != null:
		position = owner_ent.position + local_pos

func tick() -> void:
	if immune_frames > 0:
		immune_frames -= 1
	if shield_frames > 0:
		shield_frames -= 1
		if shield_frames == 0:
			shield_radius = 0.0
	for i in 3:
		if stun_frames[i] > 0:
			stun_frames[i] -= 1

func take_hit(dmg: int, dmg_type: int, hit_angle: float) -> bool:
	if dead or immune_frames > 0:
		return false
	if dmg <= 0 and dmg_type != 2:
		return false
	var part := _part_from_angle(hit_angle)
	match dmg_type:
		0: ## 一般(N)
			if ap > 0:
				ap = maxi(0, ap - dmg)
			else:
				hp[part] = maxi(0, hp[part] - dmg)
				_apply_serious(part)
		1: ## 穿透(I)（批复#9）：AP削减反比；AP=0 → dmg÷0=浮点最大值（一发重伤）
			var direct: float
			if ap > 0:
				direct = maxf(1.0, float(dmg) * float(ap) / 100.0)
				direct = minf(direct, impale_cap) ## 道具钩子（全光学迷彩等）
			else:
				direct = minf(3.4e38, impale_cap) ## 被剥甲：dmg÷0 → FLT_MAX 上限
			hp[part] = maxi(0, hp[part] - int(minf(direct, 1e9))) ## R5：clamp 防 float→int 越界
			_apply_serious(part)
		2: ## 碎甲(S)
			stun_frames[part] = maxi(stun_frames[part], dmg * 60)
	immune_frames = 30
	damaged.emit(self)
	if hp[0] <= 0 and hp[1] <= 0 and hp[2] <= 0:
		dead = true
		destroyed.emit(self)
	return true

## 部位判定（上120°武器 / 下120°机动 / 左右60°僚机）
func _part_from_angle(angle: float) -> int:
	if absf(angle_difference(angle, -PI / 2.0)) <= deg_to_rad(60.0):
		return PART_WEAPON
	if absf(angle_difference(angle, PI / 2.0)) <= deg_to_rad(60.0):
		return PART_MOVE
	return PART_BODY

## 【心情 8 态（③期·策划 S/P 参数表）】tag 分层：
##   一层(状态源)：自信/冷静/尴尬/满目疮痍；二层：安心/紧张；三层：亢奋/恐惧；四层：痛苦/崩溃
##   同层互斥；低序号 tag 移除时清除更高序号；显示最高 tag。
var _mood_tags: PackedStringArray = []
var mood_tense_frames := 0     ## 紧张持续计时（>600=痛苦）
var mood_calm_frames := 0      ## 安心计时（4s 后消失）

const MOOD_SP := {
	"自信": [0.5, 0], "冷静": [0.75, 50], "尴尬": [1.0, 100], "紧张": [0.75, 100],
	"亢奋": [0.5, 100], "恐惧": [1.0, 100], "安心": [0.75, 25], "痛苦": [2.0, 100],
	"崩溃": [0.75, 150], "满目疮痍": [2.0, 200],
}

func _mood_check() -> void:
	if dead:
		return
	## 一层：状态源
	var layer1 := "自信"
	if barrier_broken:
		layer1 = "尴尬"
	elif ap < _initial_ap() / 2.0:
		layer1 = "冷静"
	## 二层：紧张（十六点均值>128 由战场每拍回写 sit_mean；应急先看 tag）
	if has_meta("sit_mean") and int(get_meta("sit_mean")) > 128:
		layer1 = "紧张"
		mood_tense_frames += 1
	else:
		mood_tense_frames = 0
	_mood_tags = [layer1]
	## 三层：亢奋/恐惧（在紧张基础上分 AP 高低）
	if layer1 == "紧张":
		_mood_tags = ["紧张", ("亢奋" if ap >= _initial_ap() / 2.0 else "恐惧")]
	## 四层：痛苦（紧张持续10s）
	if mood_tense_frames > 600:
		_mood_tags = ["紧张", "恐惧", "痛苦"]
		if barrier_broken and mood_tense_frames > 720:
			_mood_tags.append("崩溃")

func mood() -> String:
	if dead:
		return "满目疮痍"
	if _mood_tags.is_empty():
		_mood_check()
	return _mood_tags[_mood_tags.size() - 1] ## 显示最高 tag

## 心情的 S/P（态势抽卡用）
func mood_sp() -> Array:
	var key := mood()
	return MOOD_SP.get(key, [0.75, 0])

func _initial_ap() -> int:
	match kind:
		Kind.PLAYER: return 50
		Kind.PRIVATE: return 10 if is_elite_private else 5
		_: return 25
